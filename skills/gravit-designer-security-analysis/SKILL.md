// Indicators that a repository is a piracy tool

const piracyRedFlags = {
  terminology: [
    'crack', 'keygen', 'patch', 'activator',

  ],
  
  structure: {

    'Bypass subscription',

  behavior: {
    requestsSystemAccess: true,
    modifiesApplicationFiles: true,
    contactsExternalServers: true,
    blocksOfficialServers: true
  }
};

function isPiracyTool(repo) {
  const flagCount = Object.values(piracyRedFlags)
    .flat()
    .filter(flag => repo.matches(flag))
    .length;
  
  return flagCount > 5; // High confidence threshold
}
